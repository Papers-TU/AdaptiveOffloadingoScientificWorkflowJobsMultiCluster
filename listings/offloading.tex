\begin{algorithm}[t]
\caption{Job Offloading via Makespan Prediction}
\label{alg:offloading}
\small
\SetKwFunction{Sim}{Simulate}
\SetKwProg{Fn}{Function}{:}{}
\KwIn{jobs $J$, target makespan $D$, free resources $R_\mathrm{p}$, $R_\mathrm{s}$ of primary (p) and secondary (s), criterion $\mathcal{C}$: runtime-centric (LJF), data-centric (SISF), or naive (PEFO)}
\KwOut{offloaded jobs $J_\mathrm{off}$}
$J_\mathrm{off}\gets\emptyset$\;
$\hat M\gets\Sim{$J_\mathrm{off}$}$\;
\While{$\hat M > D$ \textbf{and} $J_\mathrm{off}\neq J$}{
  add the first job of $J\setminus J_\mathrm{off}$ selected by $\mathcal{C}$ to $J_\mathrm{off}$\;
  $\hat M\gets\Sim{$J_\mathrm{off}$}$\;
}
\Return $J_\mathrm{off}$\;
\BlankLine
\Fn{\Sim{$J_\mathrm{off}$}}{
  $\hat M\gets 0$\;
  \While{unfinished jobs in $J$}{
    \ForEach{ready job $j$}{
      $c\gets\mathrm{s}$ \textbf{if} $j\in J_\mathrm{off}$ \textbf{else} $\mathrm{p}$\;
      \lIf{$\mathbf r(j)\le R_c$}{run $j$ on $c$ for $\hat t_q(j)$}
      \lElseIf{\textnormal{PEFO} \textbf{and} $\mathbf r(j)\le R_\mathrm{s}$}{run $j$ on s}
      \lElse{defer $j$}
    }
    advance $\hat M$ to the next job completion\;
  }
  \Return $\hat M$\;
}
\end{algorithm}
